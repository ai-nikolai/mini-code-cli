# Optional dependencies for the benchmark harness (benchmarks/).
#
# Deliberately NOT part of the mini-code-cli install: the CLI must stay a small,
# dependency-free package, and nobody running `mini-code` should have to download
# a benchmark harness. Install with:
#
#     python -m pip install -r benchmarks/requirements.txt
#
# Python 3.12+ is needed by terminal-bench itself (0.2.x); SWE-bench and the
# agents' CLIs work on 3.10+.

# --- benchmark harnesses -----------------------------------------------------
# Harbor: the official harness for Terminal-Bench 2.x (2.0 / 2.1), wrapped by
# benchmarks/suites/terminal_bench.py with options.harness="harbor" (the default).
# Installed straight from our fork's branch, which is where the Docker-free
# (Apptainer) backend lives -- no patch step, no local checkout required:
harbor @ git+https://github.com/ai-nikolai/harbor.git@mini-code/apptainer
# Terminal-Bench 1.x ("terminal-bench-core" datasets), wrapped with
# options.harness="tb". Optional: only needed for the older task sets.
#
# The 2.1 tasks are the part pip cannot install. ai-nikolai/terminal-bench-2-1
# is a data repository -- 89 task directories, no setup.py or pyproject.toml --
# so it has to be cloned, which is what benchmarks/scripts/setup_external_repos.sh
# does:
#
#     benchmarks/scripts/setup_external_repos.sh
#     -> external_repos/terminal-bench-2-1 @ mini-code/bench
#
# `pip install "terminal-bench @ git+https://github.com/ai-nikolai/terminal-bench-2-1.git@mini-code/bench"`
# fails with "does not appear to be a Python project: neither 'setup.py' nor
# 'pyproject.toml' found", and the 2.1 config's dataset_dir points at that clone.
terminal-bench>=0.2.18
# SWE-bench 5.0.2: `python -m swebench.harness.run_evaluation`, wrapped by
# benchmarks/suites/swebench.py. Pulls `datasets` and the Docker SDK.
swebench>=5.0.2

# --- containers --------------------------------------------------------------
docker>=7.0

# --- local model serving (H200) ----------------------------------------------
# vLLM serves the OpenAI-compatible endpoint; LiteLLM provides the
# Anthropic-compatible /v1/messages that Claude Code requires.
vllm>=0.6
litellm>=1.50
